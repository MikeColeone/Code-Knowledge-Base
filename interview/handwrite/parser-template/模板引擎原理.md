```js
// define
(function (window) {
    function fn(str) {
        this.str = str;
    }

    fn.prototype.format = function () {
        var arg = __1__;
        return this.str.replace(__2__, function (a, b) {
            return arg[b] || '';
        });
    };

    window.fn = fn;
})(window);

// use
(function () {
    var t = new fn('[{1}](https://apaas.my.feishuapp.cn/ae/apps/Interview_Question_Community__c/%7B0%7D){2}');
    console.log(t.format('http://www.alibaba.com', 'Alibaba', 'Welcome'));
})();
```

- 简单考虑模板引擎原理：
  define部分定义一个简单的模板类，使用{}作为转义标记，中间的数字表示替换目标，format实参用来替换模板内标记 横线处填：

1. Array.prototype.slice.call(arguments, 0)
2. /\{\s*(\d+)\s*\}/g
