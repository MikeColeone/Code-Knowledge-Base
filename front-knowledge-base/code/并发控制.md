async function parallel(tasks, concurrency) {
    const resolves = []
    const rejects = []
    const running = []
    for (const task of tasks){
        const p = Promise.resolve(task()).then(res=>{
            resolves.push(res)
        }).catch(err=>{
            rejects.push(err)
        }).finally(()=>running.delete(p))

        running.add(p)
        if(running.size>=concurrency){
            await Promise.race(running)
        }
    }

    await Promise.all(running)
    return {
        resolved: resolves,
        rejected: rejects
    }
}
