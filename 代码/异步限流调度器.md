- 实现JS限流调度器，方法add接收一个返回Promise的函数，同时并行处理的任务数不超过设定的上限。


```js
class Scheduler {
  limit = 2;
  taskList = [];
  currentTaskLen = 0
  constructor(options) {
    this.limit = options.limit
  }
  add(fn) {
    const wrappedFn = async () => {
      this.currentTaskLen++
      return fn().then(() => {
        this.currentTaskLen--
        this.runNext()
      })
    }
    if (this.currentTaskLen < this.limit) {
      wrappedFn()
    } else {
      this.taskList.push(wrappedFn)
    }
  }
  runNext() {
    if (!this.taskList.length) return
    const fn = this.taskList.shift()
    return fn()
  }
}
```
