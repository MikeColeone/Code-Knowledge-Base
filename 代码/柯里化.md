实现如下函数，要求返回都等于9


```js
function add() {
    const context = {
        cache: [...arguments],
    };
    function addWithCache() {
        const {
            cache = [],
        } = this;
        const args = [...arguments];
        const adder = (sum, next) => sum + next;
        if (args.length > 0) {
            for(let i = 0; i < args.length; i += 1) {
                if (typeof args[i] === 'number') {
                    cache.push(args[i]);
                } else {
                    return cache.reduce(adder);
                }
            }
        } else {
            return cache.reduce(adder) || 0;
        }
        return addWithCache.bind(this);
    }
    return addWithCache.bind(context);
}
```
